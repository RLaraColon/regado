const int PIN_RELE = 7;
const int PIN_SENSOR = A0;
const int UMBRAL_SECO = 700; // Sensor invertido

void setup() {
  pinMode(PIN_RELE, OUTPUT);
  digitalWrite(PIN_RELE, HIGH); // Apagado
  Serial.begin(9600);
}

void loop() {
  int raw = analogRead(PIN_SENSOR);
  Serial.print("Humedad ADC: "); Serial.println(raw);

  if (raw >= UMBRAL_SECO) {
    // Suelo seco -> Activar bomba
    digitalWrite(PIN_RELE, LOW);
    Serial.println(">> BOMBA ACTIVADA [REGANDO]");
  } else {
    // Suelo húmedo -> Apagar bomba
    digitalWrite(PIN_RELE, HIGH);
    Serial.println(">> Bomba en reposo");
  }
  delay(1000);
}
